#include "AppComponent.h"
//#include "ui/MainComponent.h"


//==============================================================================
AppComponent::AppComponent()
{
#if USE_GENERIC_EDITOR
	editorUI = std::make_unique<juce::GenericAudioProcessorEditor> (audioProcessor);
#else
	// TODO: implement MainComponent class for custom UI components
	// editorUI = std::make_unique<MainComponent>();
#endif

	addAndMakeVisible (*editorUI);
	setSize (600, 400);

	processorPlayer.setProcessor (&audioProcessor);
	deviceManager.addAudioCallback (&processorPlayer);

	// Some platforms require permissions to open input channels so request that here
	const int inputChannels = audioProcessor.getMainBusNumInputChannels();
	if (inputChannels > 0
		&& juce::RuntimePermissions::isRequired (juce::RuntimePermissions::recordAudio)
		&& !juce::RuntimePermissions::isGranted (juce::RuntimePermissions::recordAudio))
	{
		juce::RuntimePermissions::request (
			juce::RuntimePermissions::recordAudio,
			[safeThis = juce::Component::SafePointer<AppComponent> (this)] (bool granted)
			{
				if (safeThis != nullptr)
					safeThis->initialiseAudio (granted
						? safeThis->audioProcessor.getMainBusNumInputChannels() : 0);
			});
	}
	else
	{
		initialiseAudio (inputChannels);
	}
}

AppComponent::~AppComponent()
{
	deviceManager.removeAudioCallback (&processorPlayer);
	processorPlayer.setProcessor (nullptr);
	deviceManager.closeAudioDevice();
}

//==============================================================================
void AppComponent::initialiseAudio (int numInputChannels)
{
	const int numOutputChannels = audioProcessor.getMainBusNumOutputChannels();

	const auto error = deviceManager.initialise (
		numInputChannels, numOutputChannels, nullptr, true);

	if (error.isNotEmpty())
		juce::Logger::writeToLog ("Unable to initialise the audio device: " + error);
}

//==============================================================================
void AppComponent::paint (juce::Graphics& g)
{
	// (Our component is opaque, so we must completely fill the background with a solid colour)
	g.fillAll (getLookAndFeel().findColour (juce::ResizableWindow::backgroundColourId));

	g.setFont (juce::FontOptions (16.0f));
	g.setColour (juce::Colours::white);
	juce::Rectangle<int> text (0, 0, getWidth(), getHeight() / 2);
	g.drawText ("Finalmente l'app AUDIO funziona!", text, juce::Justification::centred, true);
}

void AppComponent::resized()
{
	editorUI->setBounds (0, getHeight() / 2, getWidth(), getHeight() / 2);
}
