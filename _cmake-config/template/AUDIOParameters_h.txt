*				'ptrs.dspModule_valuePtr = apvts.getRawParameterValue (dspModule_paramID.getParamID());'
*

#pragma once

#include <juce_audio_processors/juce_audio_processors.h>


namespace Parameters
{
	// -------------------------------------------------------------------------------- APVTS layout declaration
	juce::AudioProcessorValueTreeState::ParameterLayout createLayout();

	// -------------------------------------------------------------------------------- 1. Declare parameter IDs
	// inline const juce::ParameterID dspModule_paramID { "dspModule_ID" };		// ~~~~~~~~ reference
	inline const juce::ParameterID oscFrequency_paramID { "oscFrequency_ID" };	// ^^^^^^^^ example, disposable
	inline const juce::ParameterID gainOut_paramID { "gainOut_ID" };			// ^^^^^^^^ example, disposable

	// -------------------------------------------------------------------------------- 2. Declare raw pointers
	// One borrowed value pointer per DSP parameter.
	// Engine forwards the resolved struct to DspController, which distributes
	// the pointers to modules. Use APVTS raw pointers for reading only; change
	// parameters through JUCE parameter APIs or attachments.
	struct RawPointers
	{
		// std::atomic<float>* dspModule_valuePtr = nullptr;				// ~~~~~~~~ reference
		std::atomic<float>* oscFrequency_valuePtr = nullptr;				// ^^^^^^^^ example, disposable
		std::atomic<float>* gainOut_valuePtr = nullptr;						// ^^^^^^^^ example, disposable
	};

	RawPointers resolve (const juce::AudioProcessorValueTreeState& apvts);
}
