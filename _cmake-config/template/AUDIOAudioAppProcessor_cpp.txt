#include "AudioAppProcessor.h"


juce::AudioProcessor::BusesProperties AudioAppProcessor::createBusesProperties()
{
	auto buses = BusesProperties{};

#if KINKOUT_APP_DEFAULT_AUDIO_INPUT_CHANNELS > 0
	buses = buses.withInput (
		"Input",
		juce::AudioChannelSet::canonicalChannelSet
		(KINKOUT_APP_DEFAULT_AUDIO_INPUT_CHANNELS),
		true);
#endif

#if KINKOUT_APP_DEFAULT_AUDIO_OUTPUT_CHANNELS > 0
	buses = buses.withOutput (
		"Output",
		juce::AudioChannelSet::canonicalChannelSet
		(KINKOUT_APP_DEFAULT_AUDIO_OUTPUT_CHANNELS),
		true);
#endif

	return buses;
}

AudioAppProcessor::AudioAppProcessor()
	: AudioProcessor (createBusesProperties())
{
	oscillator.initialise ([] (float angle) { return std::sin (angle); }, oscillatorTableSize);
	oscillator.setFrequency (testFrequency);
}

AudioAppProcessor::~AudioAppProcessor() = default;

// ================================================================================ Main processor
void AudioAppProcessor::prepareToPlay (double sampleRate, int samplesPerBlock)
{
	// Use this method as the place to do any pre-playback
	// initialisation that you need..

	const juce::dsp::ProcessSpec processSpec
	{
		sampleRate,
		static_cast<juce::uint32> (samplesPerBlock),
		oscillatorChannels
	};

	oscillator.prepare (processSpec);
}

void AudioAppProcessor::processBlock (juce::AudioBuffer<float>& buffer,
									  juce::MidiBuffer& midiMessages)
{
	juce::ignoreUnused (midiMessages);

	juce::ScopedNoDenormals noDenormals;
	auto totalNumInputChannels  = getTotalNumInputChannels();
	auto totalNumOutputChannels = getTotalNumOutputChannels();

	const auto bufferNumSamples = buffer.getNumSamples();

	// In case we have more outputs than inputs, this code clears any output
	// channels that didn't contain input data, (because these aren't
	// guaranteed to be empty - they may contain garbage).
	// This is here to avoid people getting screaming feedback
	// when they first compile the application, but obviously you don't need to keep
	// this code if your algorithm always overwrites all the output channels.
	for (auto i = totalNumInputChannels; i < totalNumOutputChannels; ++i)
		buffer.clear (i, 0, bufferNumSamples);

	// This is the place where you'd normally do the guts of your application
	// audio processing...
	// Make sure to reset the state if your inner loop is processing
	// the samples and the outer loop is handling the channels.
	// Alternatively, you can process the samples with the channels
	// interleaved by keeping the same state.

	// Generate the test signal only when an output is available, while allowing
	// input-only applications to continue processing the incoming audio.
	if (totalNumOutputChannels > 0)
	{
		auto* channelData = buffer.getWritePointer (0);

		for (int sample = 0; sample < bufferNumSamples; ++sample)
			channelData[sample] = oscillator.processSample (0.0f) * testLevel;

		for (int channel = 1; channel < totalNumOutputChannels; ++channel)
			buffer.copyFrom (channel, 0, buffer, 0, 0, bufferNumSamples);
	}
}

void AudioAppProcessor::releaseResources()
{
	oscillator.reset();
}

bool AudioAppProcessor::isBusesLayoutSupported (const BusesLayout&) const
{
	return true;
}

// ================================================================================ Processor info

const juce::String AudioAppProcessor::getName() const { return JUCE_APPLICATION_NAME_STRING; }

double AudioAppProcessor::getTailLengthSeconds() const
{
	return 0.0;
}

bool AudioAppProcessor::acceptsMidi() const
{
#if KINKOUT_APP_WANTS_MIDI_INPUT
	return true;
#else
	return false;
#endif
}

bool AudioAppProcessor::producesMidi() const
{
#if KINKOUT_APP_PRODUCES_MIDI_OUTPUT
	return true;
#else
	return false;
#endif
}

// ================================================================================ State

void AudioAppProcessor::getStateInformation (juce::MemoryBlock& destData)
{
	destData.setSize (0, false);
}

void AudioAppProcessor::setStateInformation (const void* data, int sizeInBytes)
{
	juce::ignoreUnused (data, sizeInBytes);
}
