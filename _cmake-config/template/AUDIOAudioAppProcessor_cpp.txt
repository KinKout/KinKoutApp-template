#include "AudioAppProcessor.h"
#include "engine/Parameters.h"


juce::AudioProcessor::BusesProperties AudioAppProcessor::createBusesProperties()
{
	// The channel counts come from appConfig.cmake as compile definitions.
	// A bus is added only when its channel count is greater than zero.
	// This supports layouts with an input bus, an output bus, or both.
	// The current DSP chain still needs an output bus to run.
	// canonicalChannelSet() selects the standard JUCE layout for that channel count.
	auto buses = BusesProperties{};

#if KINKOUT_APP_DEFAULT_AUDIO_INPUT_CHANNELS > 0
	buses = buses.withInput (
		"Input",
		juce::AudioChannelSet::canonicalChannelSet
		(KINKOUT_APP_DEFAULT_AUDIO_INPUT_CHANNELS),
		true);
#endif

#if KINKOUT_APP_DEFAULT_AUDIO_OUTPUT_CHANNELS > 0
	buses = buses.withOutput (
		"Output",
		juce::AudioChannelSet::canonicalChannelSet
		(KINKOUT_APP_DEFAULT_AUDIO_OUTPUT_CHANNELS),
		true);
#endif

	return buses;
}

AudioAppProcessor::AudioAppProcessor()
	: AudioProcessor (createBusesProperties()),
	  apvts (*this, nullptr, "PARAMETERS", Parameters::createLayout()),
	  engine (instRack)
{
	// Resolve the parameter IDs once and hand the raw pointers down the chain.
	engine.bindParameters (Parameters::resolve (apvts));
	setLatencySamples (engine.getLatencySamples());
}

AudioAppProcessor::~AudioAppProcessor() = default;

// ================================================================================ Main processor
void AudioAppProcessor::prepareToPlay (double sampleRate, int samplesPerBlock)
{
	engine.prepare (sampleRate,
		samplesPerBlock,
		getTotalNumOutputChannels());
	setLatencySamples (engine.getLatencySamples());
}

void AudioAppProcessor::processBlock (juce::AudioBuffer<float>& buffer,
									  juce::MidiBuffer& midiMessages)
{
	juce::ScopedNoDenormals noDenormals;
	const int outputChannels = getTotalNumOutputChannels();
	const int inputChannels = getTotalNumInputChannels();

	for (int channel = inputChannels; channel < outputChannels; ++channel)
		buffer.clear (channel, 0, buffer.getNumSamples());

	// This template uses an oscillator followed by a gain.
	// The DSP chain runs only when at least one output channel is available.
	// getBusBuffer() gets the first output bus. Engine processes its buffer.
	if (outputChannels > 0)
	{
		auto output = getBusBuffer (buffer, false, 0);
		engine.process (&output, nullptr, &midiMessages);
	}
}

void AudioAppProcessor::releaseResources()
{
	// When playback stops, you can use this as an opportunity to free up any
	// spare memory, etc.

	engine.reset();
}

bool AudioAppProcessor::isBusesLayoutSupported (const BusesLayout&) const
{
	// AudioProcessorPlayer checks this method before it connects the processor
	// buses to the active device channels. The current oscillator and gain can
	// work with any output channel count and do not need matching input and output
	// layouts. Add layout checks here if a future DSP chain has fixed channel needs.

	return true;
}

// ================================================================================ Processor info

const juce::String AudioAppProcessor::getName() const { return JUCE_APPLICATION_NAME_STRING; }

double AudioAppProcessor::getTailLengthSeconds() const
{
	return engine.getTailLengthSeconds();
}

bool AudioAppProcessor::acceptsMidi() const
{
#if KINKOUT_APP_WANTS_MIDI_INPUT
	return true;
#else
	return false;
#endif
}

bool AudioAppProcessor::producesMidi() const
{
#if KINKOUT_APP_PRODUCES_MIDI_OUTPUT
	return true;
#else
	return false;
#endif
}
