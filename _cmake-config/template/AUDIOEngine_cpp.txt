#include "Engine.h"


Engine::Engine (InstRack& rackToUse)
	: dspController (rackToUse)
{
}

Engine::~Engine() = default;

void Engine::bindParameters (const Parameters::RawPointers& params)
{
	dspController.bindParameters (params);
}

void Engine::prepare (const double sampleRate, const int maxBlockSize, const int numChannels)
{
	dspController.prepare (sampleRate, maxBlockSize, numChannels);
}

void Engine::process	(juce::AudioBuffer<float>* buffer,
						const juce::AudioBuffer<float>* sidechain,
						juce::MidiBuffer* midi)
{
	dspController.process (buffer, sidechain, midi);
}

void Engine::reset()
{
	dspController.reset();
}

int Engine::getLatencySamples() const
{
	return dspController.getLatencySamples();
}

double Engine::getTailLengthSeconds() const
{
	return dspController.getTailLengthSeconds();
}
